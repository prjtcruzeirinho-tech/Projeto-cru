const FormularioInscricao = ({ user, showToast, categoriaPreSelecionada }) => {
  const [formData, setFormData] = useState({
    nome: '', nascimento: '', responsavel: '', whatsapp: '', categoria: categoriaPreSelecionada, tipo: 'Normal'
  });
  const [loading, setLoading] = useState(false);

  useEffect(() => { setFormData(prev => ({...prev, categoria: categoriaPreSelecionada})); }, [categoriaPreSelecionada]);

  const handleDateMask = (e) => {
    let v = e.target.value.replace(/\D/g, '');
    if (v.length > 2) v = v.slice(0, 2) + '/' + v.slice(2);
    if (v.length > 5) v = v.slice(0, 5) + '/' + v.slice(5);
    setFormData({ ...formData, nascimento: v.slice(0, 10) });
  };

  const handlePhoneMask = (e) => {
    let v = e.target.value.replace(/\D/g, '');
    if (v.length > 2) v = '(' + v.slice(0, 2) + ') ' + v.slice(2);
    if (v.length > 9) v = v.slice(0, 10) + '-' + v.slice(10);
    setFormData({ ...formData, whatsapp: v.slice(0, 15) });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!user) return showToast('Erro de conexão. Aguarde.', 'error');
    if (formData.nascimento.length !== 10 || formData.whatsapp.length < 14) return showToast('Preencha os dados corretamente.', 'error');
    setLoading(true);
    try {
      await addDoc(getPublicCollection('inscricoes'), { ...formData, status: 'Pendente', dataEnvio: new Date().toISOString() });
      showToast('Inscrição enviada com sucesso!');
      setFormData({ nome: '', nascimento: '', responsavel: '', whatsapp: '', categoria: categoriaPreSelecionada, tipo: 'Normal' });
    } catch (err) {
      showToast('Erro ao salvar inscrição.', 'error');
    }
    setLoading(false);
  };

  return (
    <div className="max-w-xl mx-auto bg-white p-8 rounded-2xl shadow-xl border border-gray-100 mt-6">
      <h2 className="text-3xl font-bold text-blue-900 mb-6 text-center">Inscrição {categoriaPreSelecionada}</h2>
      <form onSubmit={handleSubmit} className="space-y-5">
        <div>
          <label className="block text-sm font-semibold text-gray-700 mb-1">Nome Completo do Atleta</label>
          <input required type="text" value={formData.nome} onChange={e => setFormData({...formData, nome: e.target.value})} className="w-full p-3 rounded-lg border focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Ex: Enzo Gabriel" />
        </div>
        <div>
          <label className="block text-sm font-semibold text-gray-700 mb-1">Data de Nascimento</label>
          <input required type="text" value={formData.nascimento} onChange={handleDateMask} placeholder="DD/MM/AAAA" className="w-full p-3 rounded-lg border focus:ring-2 focus:ring-blue-500 outline-none" />
        </div>
        <div>
          <label className="block text-sm font-semibold text-gray-700 mb-1">Nome do Responsável</label>
          <input required type="text" value={formData.responsavel} onChange={e => setFormData({...formData, responsavel: e.target.value})} className="w-full p-3 rounded-lg border focus:ring-2 focus:ring-blue-500 outline-none" />
        </div>
        <div>
          <label className="block text-sm font-semibold text-gray-700 mb-1">WhatsApp</label>
          <input required type="text" value={formData.whatsapp} onChange={handlePhoneMask} placeholder="(15) 99999-9999" className="w-full p-3 rounded-lg border focus:ring-2 focus:ring-blue-500 outline-none" />
        </div>
        <div>
          <label className="block text-sm font-semibold text-gray-700 mb-1">Tipo de Inscrição</label>
          <select value={formData.tipo} onChange={e => setFormData({...formData, tipo: e.target.value})} className="w-full p-3 rounded-lg border focus:ring-2 focus:ring-blue-500 outline-none">
            <option value="Normal">Normal (R$ 25)</option>
            <option value="Solidária">Solidária (Isento - Sujeito a Análise)</option>
          </select>
        </div>
        <button disabled={loading} type="submit" className="w-full bg-blue-600 hover:bg-blue-700 text-white font-bold py-4 rounded-lg shadow-md transition-colors disabled:opacity-50">
          {loading ? 'Enviando...' : 'Confirmar Inscrição'}
        </button>
      </form>
    </div>
  );
};

const ClassificacaoPublic = ({ partidas, equipes }) => {
  const [categoria, setCategoria] = useState('SUB-9');

  const tabela = useMemo(() => {
    let tab = {};
    equipes.filter(e => e.categoria === categoria).forEach(eq => {
      tab[eq.id] = { id: eq.id, nome: eq.nome, cor: eq.corPrincipal, pts: 0, j: 0, v: 0, e: 0, d: 0, gp: 0, gc: 0, sg: 0 };
    });

    partidas.filter(p => p.categoria === categoria && p.status === 'Finalizado').forEach(p => {
      const pm = parseInt(p.placarMandante) || 0;
      const pv = parseInt(p.placarVisitante) || 0;
      
      if(tab[p.mandanteId]) {
        tab[p.mandanteId].j += 1;
        tab[p.mandanteId].gp += pm;
        tab[p.mandanteId].gc += pv;
        tab[p.mandanteId].sg += (pm - pv);
        if(pm > pv) { tab[p.mandanteId].v += 1; tab[p.mandanteId].pts += 3; }
        else if (pm === pv) { tab[p.mandanteId].e += 1; tab[p.mandanteId].pts += 1; }
        else { tab[p.mandanteId].d += 1; }
      }

      if(tab[p.visitanteId]) {
        tab[p.visitanteId].j += 1;
        tab[p.visitanteId].gp += pv;
        tab[p.visitanteId].gc += pm;
        tab[p.visitanteId].sg += (pv - pm);
        if(pv > pm) { tab[p.visitanteId].v += 1; tab[p.visitanteId].pts += 3; }
        else if (pv === pm) { tab[p.visitanteId].e += 1; tab[p.visitanteId].pts += 1; }
        else { tab[p.visitanteId].d += 1; }
      }
    });

    return Object.values(tab).sort((a, b) => b.pts - a.pts || b.v - a.v || b.sg - a.sg || b.gp - a.gp);
  }, [partidas, equipes, categoria]);

  return (
    <div className="max-w-5xl mx-auto mt-6 p-4">
      <div className="flex justify-between items-center mb-6">
        <h2 className="text-3xl font-bold text-blue-900 flex items-center"><Trophy className="mr-2"/> Classificação</h2>
        <select value={categoria} onChange={e=>setCategoria(e.target.value)} className="p-2 border rounded shadow-sm text-lg font-bold text-blue-800">
          <option value="SUB-9">SUB-9</option><option value="SUB-11">SUB-11</option><option value="SUB-16">SUB-16</option>
        </select>
      </div>
      <div className="bg-white rounded-xl shadow-lg overflow-x-auto">
        <table className="w-full text-left min-w-[600px]">
          <thead className="bg-blue-900 text-white">
            <tr>
              <th className="p-4 rounded-tl-xl text-center w-12">#</th>
              <th className="p-4">Equipe</th>
              <th className="p-4 text-center">PTS</th>
              <th className="p-4 text-center">J</th>
              <th className="p-4 text-center">V</th>
              <th className="p-4 text-center">E</th>
              <th className="p-4 text-center">D</th>
              <th className="p-4 text-center">GP</th>
              <th className="p-4 text-center">GC</th>
              <th className="p-4 text-center rounded-tr-xl">SG</th>
            </tr>
          </thead>
          <tbody>
            {tabela.map((t, idx) => (
              <tr key={t.id} className="border-b hover:bg-gray-50">
                <td className="p-4 text-center font-bold">{idx + 1}º</td>
                <td className="p-4 font-bold flex items-center">
                  <div className="w-4 h-4 rounded-full mr-2" style={{backgroundColor: t.cor}}></div>
                  {t.nome}
                </td>
                <td className="p-4 text-center font-black text-blue-700">{t.pts}</td>
                <td className="p-4 text-center">{t.j}</td>
                <td className="p-4 text-center">{t.v}</td>
                <td className="p-4 text-center">{t.e}</td>
                <td className="p-4 text-center">{t.d}</td>
                <td className="p-4 text-center">{t.gp}</td>
                <td className="p-4 text-center">{t.gc}</td>
                <td className="p-4 text-center font-semibold">{t.sg}</td>
              </tr>
            ))}
            {tabela.length === 0 && <tr><td colSpan="10" className="p-6 text-center text-gray-500">Nenhuma equipe nesta categoria.</td></tr>}
          </tbody>
        </table>
      </div>
    </div>
  );
};

const AdminInscricoes = ({ inscricoes, showToast }) => {
  const handleStatus = async (id, status, dados) => {
    try {
      await updateDoc(getPublicDoc('inscricoes', id), { status });
      if (status === 'Aprovada') {
        // Criar atleta
        await addDoc(getPublicCollection('atletas'), {
          nome: dados.nome, categoria: dados.categoria, responsavel: dados.responsavel, 
          tipo: dados.tipo, equipeId: null, gols: 0, assistencias: 0
        });
      }
      showToast(`Inscrição ${status.toLowerCase()}!`);
    } catch (e) { showToast('Erro', 'error'); }
  };

  const handleExcluir = async (id) => {
    if(!window.confirm("Excluir inscrição permanentemente?")) return;
    try {
      await deleteDoc(getPublicDoc('inscricoes', id));
      showToast('Registro excluído!');
    } catch (e) { showToast('Erro', 'error'); }
  }

  return (
    <div className="bg-white rounded-xl shadow p-6 overflow-x-auto">
      <h2 className="text-2xl font-bold text-gray-800 mb-4">Gerenciar Inscrições</h2>
      <table className="w-full text-left border-collapse min-w-[700px]">
        <thead>
          <tr className="bg-gray-100 text-gray-700 uppercase text-sm">
            <th className="p-3">Atleta</th>
            <th className="p-3">Categoria</th>
            <th className="p-3">Tipo</th>
            <th className="p-3">Status</th>
            <th className="p-3">Ações</th>
          </tr>
        </thead>
        <tbody>
          {inscricoes.map(i => (
            <tr key={i.id} className="border-b hover:bg-gray-50">
              <td className="p-3 font-semibold">{i.nome}<br/><span className="text-xs font-normal text-gray-500">{i.nascimento} | {i.responsavel}</span></td>
              <td className="p-3">{i.categoria}</td>

            </tr>

        </tbody>
      </table>
    </div>
  );
};

const AdminSorteio = ({ atletas, equipes, showToast }) => {
  const [categoria, setCategoria] = useState('SUB-9');
  const [sorteando, setSorteando] = useState(false);

  const atletasLivres = atletas.filter(a => a.categoria === categoria && !a.equipeId);
  const equipesCat = equipes.filter(e => e.categoria === categoria);

  const realizarSorteio = async () => {
    if (equipesCat.length === 0) return showToast('Cadastre equipes nesta categoria primeiro!', 'error');
    if (atletasLivres.length === 0) return showToast('Nenhum atleta aprovado sem equipe nesta categoria.', 'error');
    if (!window.confirm(`Sortear ${atletasLivres.length} atletas para ${equipesCat.length} equipes?`)) return;

    setSorteando(true);
    try {
      // Fisher-Yates Shuffle
      let embaralhados = [...atletasLivres];
      for (let i = embaralhados.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [embaralhados[i], embaralhados[j]] = [embaralhados[j], embaralhados[i]];
      }

      // Distribuição round-robin
      let equipeIdx = 0;
      for (const atleta of embaralhados) {
        const eqId = equipesCat[equipeIdx].id;
        await updateDoc(getPublicDoc('atletas', atleta.id), { equipeId: eqId });
        equipeIdx = (equipeIdx + 1) % equipesCat.length;
      }
      showToast('Sorteio realizado e salvo com sucesso!');
    } catch (e) {
      showToast('Erro ao realizar sorteio.', 'error');
    }
    setSorteando(false);
  };

  return (
    <div className="bg-white rounded-xl shadow p-6">
      <h2 className="text-2xl font-bold text-gray-800 mb-6 flex items-center"><Shuffle className="mr-2"/> Sorteio de Atletas</h2>
      
      <div className="flex flex-col md:flex-row gap-4 items-center mb-8 p-4 bg-gray-50 rounded-lg">
        <label className="font-bold text-gray-700">Categoria:</label>
        <select value={categoria} onChange={e=>setCategoria(e.target.value)} className="p-2 border rounded font-bold text-blue-800">
          <option value="SUB-9">SUB-9</option><option value="SUB-11">SUB-11</option><option value="SUB-16">SUB-16</option>
        </select>
        <div className="flex-1 text-center md:text-right">
          <span className="text-sm bg-yellow-100 text-yellow-800 px-3 py-1 rounded-full mr-2 font-semibold">Atletas Livres: {atletasLivres.length}</span>
          <span className="text-sm bg-blue-100 text-blue-800 px-3 py-1 rounded-full font-semibold">Equipes: {equipesCat.length}</span>
        </div>
        <button disabled={sorteando || atletasLivres.length===0 || equipesCat.length===0} onClick={realizarSorteio} className="bg-blue-600 text-white font-bold py-2 px-6 rounded-lg hover:bg-blue-700 disabled:opacity-50 flex items-center">
          <Shuffle size={18} className="mr-2"/> {sorteando ? 'Sorteando...' : 'Sortear Agora'}
        </button>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
        {equipesCat.map(eq => {
          const elenco = atletas.filter(a => a.equipeId === eq.id);
          return (
            <div key={eq.id} className="border rounded-lg overflow-hidden shadow-sm">
              <div className="p-3 text-white font-bold flex justify-between" style={{backgroundColor: eq.corPrincipal}}>
                <span>{eq.nome}</span> <span>{elenco.length}</span>
              </div>
              <ul className="p-3 text-sm space-y-1 bg-white min-h-[100px]">
                {elenco.map(a => (
                  <li key={a.id} className="flex justify-between items-center border-b pb-1">
                    <span>{a.nome}</span>
                    <button onClick={()=>updateDoc(getPublicDoc('atletas', a.id), {equipeId: null})} className="text-red-500 hover:text-red-700" title="Remover da equipe"><X size={14}/></button>
                  </li>
                ))}
                {elenco.length===0 && <li className="text-gray-400 italic">Nenhum atleta</li>}
              </ul>
            </div>
          )
        })}
      </div>
    </div>
  );
};

const AdminPartidas = ({ partidas, equipes, showToast }) => {
  const [novoJogo, setNovoJogo] = useState({ categoria: 'SUB-9', data: '', horario: '', mandanteId: '', visitanteId: '' });

  const equipesCat = equipes.filter(e => e.categoria === novoJogo.categoria);

  const handleCriar = async (e) => {
    e.preventDefault();
    if(novoJogo.mandanteId === novoJogo.visitanteId) return showToast('Escolha equipes diferentes', 'error');
    if(!novoJogo.mandanteId || !novoJogo.visitanteId) return showToast('Selecione os dois times', 'error');
    
    try {
      await addDoc(getPublicCollection('partidas'), { ...novoJogo, status: 'Agendado', placarMandante: 0, placarVisitante: 0 });
      showToast('Partida criada!');
      setNovoJogo({...novoJogo, data: '', horario: '', mandanteId: '', visitanteId: ''});
    } catch(err) { showToast('Erro', 'error'); }
  };

  const handleFinalizar = async (id, pm, pv) => {
    try {
      await updateDoc(getPublicDoc('partidas', id), { status: 'Finalizado', placarMandante: pm, placarVisitante: pv });
      showToast('Partida finalizada. Classificação atualizada!');
    } catch(err) { showToast('Erro', 'error'); }
  };

  const handleExcluir = async (id) => {
    if(!window.confirm("Excluir partida?")) return;
    try { await deleteDoc(getPublicDoc('partidas', id)); showToast('Excluída!'); } catch(err) { showToast('Erro', 'error'); }
  };

  return (
    <div className="space-y-6">
      <div className="bg-white rounded-xl shadow p-6">
        <h2 className="text-xl font-bold mb-4 flex items-center"><Plus className="mr-2"/> Agendar Partida</h2>
        <form onSubmit={handleCriar} className="grid grid-cols-1 md:grid-cols-6 gap-4 items-end">

            <label className="block text-sm font-semibold mb-1">Categoria</label>

            </select>
          </div>
          <div>
            <label className="block text-sm font-semibold mb-1">Data</label>
            <input required type="date" value={novoJogo.data} onChange={e=>setNovoJogo({...novoJogo, data: e.target.value})} className="w-full p-2 border rounded"/>
          </div>
          <div>
            <label className="block text-sm font-semibold mb-1">Hora</label>
            <input required type="time" value={novoJogo.horario} onChange={e=>setNovoJogo({...novoJogo, horario: e.target.value})} className="w-full p-2 border rounded"/>
          </div>
          <div>
            <label className="block text-sm font-semibold mb-1">Mandante</label>
            <select required value={novoJogo.mandanteId} onChange={e=>setNovoJogo({...novoJogo, mandanteId: e.target.value})} className="w-full p-2 border rounded">
              <option value="">Selecione</option>
              {equipesCat.map(e => <option key={e.id} value={e.id}>{e.nome}</option>)}
            </select>
          </div>
          <div>
            <label className="block text-sm font-semibold mb-1">Visitante</label>
            <select required value={novoJogo.visitanteId} onChange={e=>setNovoJogo({...novoJogo, visitanteId: e.target.value})} className="w-full p-2 border rounded">
              <option value="">Selecione</option>
              {equipesCat.map(e => <option key={e.id} value={e.id}>{e.nome}</option>)}
            </select>
          </div>
          <button type="submit" className="bg-blue-600 text-white font-bold p-2 rounded hover:bg-blue-700">Salvar</button>
        </form>
      </div>

      <div className="bg-white rounded-xl shadow p-6 overflow-x-auto">
        <h2 className="text-xl font-bold mb-4">Gerenciar Resultados</h2>
        <table className="w-full text-left min-w-[700px]">
          <thead>
            <tr className="bg-gray-100">
              <th className="p-2">Data/Hora</th>
              <th className="p-2">Cat</th>
              <th className="p-2 text-right">Mandante</th>
              <th className="p-2 text-center">Placar</th>
              <th className="p-2 text-left">Visitante</th>
              <th className="p-2">Status</th>
              <th className="p-2">Ações</th>
            </tr>
          </thead>
          <tbody>
            {partidas.map(p => {
              const m = equipes.find(e=>e.id===p.mandanteId);
              const v = equipes.find(e=>e.id===p.visitanteId);

                <tr key={p.id} className="border-b">
                  <td className="p-2 text-sm">{p.data} {p.horario}</td>

                  </td>
                </tr>
              )
            })}
          </tbody>
        </table>
      </div>
    </div>
  );
};

const AdminEstatisticas = ({ atletas, equipes, showToast }) => {
  const [busca, setBusca] = useState('');

  const updateStats = async (id, campo, delta) => {
    const atleta = atletas.find(a => a.id === id);
    let newVal = (atleta[campo] || 0) + delta;
    if (newVal < 0) newVal = 0;
    try {
      await updateDoc(getPublicDoc('atletas', id), { [campo]: newVal });
      showToast('Estatística atualizada!');
    } catch(err) { showToast('Erro', 'error'); }
  };

  const filtrados = atletas.filter(a => a.nome.toLowerCase().includes(busca.toLowerCase()) && a.equipeId);

  return (
    <div className="bg-white rounded-xl shadow p-6">
      <h2 className="text-2xl font-bold text-gray-800 mb-4">Lançar Gols e Assistências</h2>
      <input type="text" placeholder="Buscar atleta por nome..." value={busca} onChange={e=>setBusca(e.target.value)} className="w-full p-3 border rounded-lg mb-6"/>
      
      <div className="overflow-x-auto">
        <table className="w-full text-left">
          <thead className="bg-gray-100">
            <tr>
              <th className="p-3">Atleta</th>
              <th className="p-3">Equipe</th>
              <th className="p-3 text-center">Gols</th>
              <th className="p-3 text-center">Assistências</th>
            </tr>
          </thead>
          <tbody>
            {filtrados.slice(0, 10).map(a => {
              const eq = equipes.find(e=>e.id===a.equipeId);
              return (
                <tr key={a.id} className="border-b">
                  <td className="p-3 font-bold">{a.nome} <span className="text-xs text-gray-400">({a.categoria})</span></td>
                  <td className="p-3" style={{color: eq?.corPrincipal}}>{eq?.nome}</td>
                  <td className="p-3 text-center">
                    <button onClick={()=>updateStats(a.id, 'gols', -1)} className="px-2 bg-gray-200 rounded">-</button>
                    <span className="mx-3 font-bold text-lg">{a.gols || 0}</span>
                    <button onClick={()=>updateStats(a.id, 'gols', 1)} className="px-2 bg-green-500 text-white rounded">+</button>
                  </td>
                  <td className="p-3 text-center">
                    <button onClick={()=>updateStats(a.id, 'assistencias', -1)} className="px-2 bg-gray-200 rounded">-</button>
                    <span className="mx-3 font-bold text-lg">{a.assistencias || 0}</span>
                    <button onClick={()=>updateStats(a.id, 'assistencias', 1)} className="px-2 bg-blue-500 text-white rounded">+</button>
                  </td>
                </tr>
              )
            })}
          </tbody>
        </table>
      </div>
    </div>
  );
};

export default function App() {
  const [user, setUser] = useState(null);
  const [route, setRoute] = useState('home');
  const [isAdmin, setIsAdmin] = useState(false);
  
  // Login State
  const [adminUser, setAdminUser] = useState('');
  const [adminPass, setAdminPass] = useState('');
  
  // Dados Firebase
  const [inscricoes, setInscricoes] = useState([]);
  const [atletas, setAtletas] = useState([]);
  const [equipes, setEquipes] = useState([]);
  const [partidas, setPartidas] = useState([]);
  const [toast, setToast] = useState({ show: false, msg: '', type: '' });

  const showToast = (msg, type = 'success') => {
    setToast({ show: true, msg, type });
    setTimeout(() => setToast({ show: false, msg: '', type: '' }), 4000);
  };

  // Inicialização Auth (Regra 3)
  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (err) { console.error("Auth falhou", err); }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return () => unsubscribe();
  }, []);

  // Busca de Dados (Regras 1 e 2)
  useEffect(() => {
    if (!user) return;
    const unsubInsc = onSnapshot(getPublicCollection('inscricoes'), (snap) => setInscricoes(snap.docs.map(d => ({ id: d.id, ...d.data() }))));
    const unsubAtl = onSnapshot(getPublicCollection('atletas'), (snap) => setAtletas(snap.docs.map(d => ({ id: d.id, ...d.data() }))));
    const unsubEq = onSnapshot(getPublicCollection('equipes'), (snap) => setEquipes(snap.docs.map(d => ({ id: d.id, ...d.data() }))));
    const unsubPart = onSnapshot(getPublicCollection('partidas'), (snap) => setPartidas(snap.docs.map(d => ({ id: d.id, ...d.data() }))));
    return () => { unsubInsc(); unsubAtl(); unsubEq(); unsubPart(); };
  }, [user]);

  const handleLogin = (e) => {
    e.preventDefault();
    if (adminUser === 'admin' && adminPass === 'Cruzeiro@2026') {
      setIsAdmin(true);
      setRoute('admin_dashboard');

    } else {
      showToast('Credenciais Inválidas!', 'error');

          <div className="relative group">
            <button className="px-2 py-1 rounded text-sm hover:bg-blue-700 flex items-center">Inscrições</button>
            <div className="absolute hidden group-hover:block bg-white text-blue-900 shadow-xl rounded mt-1 overflow-hidden z-50">
              <div onClick={()=>setRoute('inscricao_sub9')} className="px-4 py-2 hover:bg-blue-50 cursor-pointer whitespace-nowrap border-b text-sm">SUB-9</div>
              <div onClick={()=>setRoute('inscricao_sub11')} className="px-4 py-2 hover:bg-blue-50 cursor-pointer whitespace-nowrap border-b text-sm">SUB-11</div>
              <div onClick={()=>setRoute('inscricao_sub16')} className="px-4 py-2 hover:bg-blue-50 cursor-pointer whitespace-nowrap text-sm">SUB-16</div>
            </div>
          </div>
          
          <button onClick={() => setRoute('tabela')} className={`px-2 py-1 rounded text-sm transition-colors ${route==='tabela'?'bg-blue-700':'hover:bg-blue-700'}`}>Tabela</button>

      { id: 'admin_partidas', label: 'Jogos e Placar', icon: <CalendarDays size={18} /> },
      { id: 'admin_stats', label: 'Artilharia/Assists', icon: <Goal size={18} /> },
      { id: 'home', label: 'Sair do Painel', icon: <LogIn size={18} /> },

    return (
      <div className="flex h-screen bg-gray-100 overflow-hidden">
        {/* Sidebar */}

            <Settings className="mr-2"/> ÁREA RESTRITA

            {menus.map(m => (

        {/* Main Content Admin */}

          <div className="flex items-center justify-center h-[80vh] p-4 animate-fade-in">
            <form onSubmit={handleLogin} className="bg-white p-8 rounded-2xl shadow-2xl max-w-sm w-full border border-gray-100">
              <div className="flex justify-center mb-6"><Shield size={48} className="text-slate-800"/></div>
              <h2 className="text-2xl font-bold text-center text-gray-800 mb-6">Acesso Restrito</h2>
              <input type="text" value={adminUser} onChange={e=>setAdminUser(e.target.value)} placeholder="Usuário" required
                className="w-full p-3 mb-3 border rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" />
              <input type="password" value={adminPass} onChange={e=>setAdminPass(e.target.value)} placeholder="Senha" required
                className="w-full p-3 mb-6 border rounded-lg focus:ring-2 focus:ring-blue-500 outline-none" />
              <button type="submit" className="w-full bg-slate-800 hover:bg-slate-900 text-white font-bold py-4 rounded-lg transition-colors">
                Entrar no Sistema
              </button>
              <button type="button" onClick={()=>setRoute('home')} className="w-full mt-4 text-gray-500 hover:text-gray-800 text-sm">Voltar ao site</button>
            </form>
          </div>
        )}

        {/* Rotas Admin */}
        {isAdmin && renderAdminLayout()}
      </main>
      
      {!isAdmin && (
        <footer className="bg-blue-900 text-blue-200 py-6 text-center text-sm">
          <p className="font-bold">Projeto Cruzeirinho — Bom de Bola, Bom na Escola</p>
          <p className="mt-1 opacity-75">Sistema de Gestão Esportiva</p>
        </footer>
      )}
    </div>
  );
}
